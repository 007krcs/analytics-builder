import type { PivotConfig } from '@analytix/core';

const config: PivotConfig = {
  id:           'revenue-pivot',
  datasetId:    'sales',
  rowFields:    ['region'],
  columnFields: ['category'],
  valueFields:  [
    { columnId: 'revenue',       aggregation: 'sum', format: 'currency' },
    { columnId: 'profit_margin',  aggregation: 'avg', format: 'percent'  },
  ],
  filters:          [],
  showRowTotals:     true,
  showColumnTotals:  true,
};
// Two row fields = region → country hierarchy
const config: PivotConfig = {
  id:           'region-country-pivot',
  datasetId:    'sales',
  rowFields:    ['region', 'country'],
  columnFields: ['quarter'],
  valueFields:  [{
    columnId:    'revenue',
    aggregation: 'sum',
    format:      'currency',
  }],
  filters:      [],
  showSubTotals: true,
  compactMode:  false,
};
// Filter to North America + Software only
const config: PivotConfig = {
  id:           'filtered-pivot',
  datasetId:    'sales',
  rowFields:    ['salesperson'],
  columnFields: ['quarter'],
  valueFields:  [{ columnId: 'revenue', aggregation: 'sum' }],
  filters: [
    { columnId: 'region',   operator: 'eq', value: 'North America' },
    { columnId: 'category', operator: 'eq', value: 'Software'       },
  ],
};

tsx BarChartExample.tsx 
import { ChartRenderer } from '@analytix/react';
import { ChartRegistry } from '@analytix/chart-engine';

const config = ChartRegistry.buildConfig('bar-grouped', {
  x:     'region',
  y:     'revenue',
  group: 'category',
});

function RevenueChart({ rows }) {
  return <ChartRenderer config={config} data={rows} height={400} />;
}

Scheduled Report Examples

ts weekly-report.ts 
import { ReportBuilder } from '@analytix/report-builder';

const report = ReportBuilder
  .create('Weekly Sales Report')
  .addText(
    'Executive Summary',
    'Weekly performance overview for the sales organisation.'
  )
  .addKpiSummary([revenueKpi, profitKpi, unitsKpi])
  .addPivot(pivotConfig)
  .addChart(chartConfig)
  .schedule({
    type:    'cron',
    cron:    '0 8 * * 1',   // Every Monday at 8am
    formats: ['pdf', 'excel'],
    enabled: true,
  })
  .build();
